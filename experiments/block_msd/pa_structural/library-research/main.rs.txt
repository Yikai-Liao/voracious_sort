use std::{hint::black_box, time::Instant};
use hakmem::isa::{Isa, Portable};
fn hak<I: Isa>(cpu: I, values: &[u64], mask: u64) -> u64 {
    values.iter().fold(0, |sum,&x| sum ^ cpu.pext(x, mask))
}
fn main() {
    let values: Vec<u64> = (0u64..1_048_576).map(|x| x.wrapping_mul(6364136223846793005)).collect();
    for q in [2,8,16,24,32] {
        let mask = (0..q).fold(0,|m,j|m|1u64<<(j*63/(q-1)));
        let mut expected = None;
        for rep in 0..4 {
            for which in 0..3 {
                let xs=black_box(values.as_slice());let m=black_box(mask);
                let start=Instant::now();
                let sum=match which {
                    0 => xs.iter().fold(0,|s,&x|s^bitkit::Bits::new(x).gather(bitkit::Bits::new(m)).get()),
                    1 => hakmem::dispatch!(|cpu|hak(cpu,xs,m)),
                    _ => hak(Portable,xs,m),
                };
                black_box(sum);
                assert_eq!(*expected.get_or_insert(sum),sum);
                println!("{q},{rep},{which},{}",start.elapsed().as_nanos());
            }
        }
    }
    println!("backend={:?}",hakmem::isa::detect());
}
